(function () {
  function panelFor(link) {
    var key = link.getAttribute('data-bibtex');
    var row = link.closest('tr');
    if (!row) return null;
    var next = row.nextElementSibling;
    if (next && next.classList.contains('bibtex-row') && next.getAttribute('data-key') === key) return next;
    var src = document.getElementById('bibtex-' + key);
    if (!src) return null;
    var tr = document.createElement('tr');
    tr.className = 'bibtex-row';
    tr.setAttribute('data-key', key);
    tr.hidden = true;
    var td = document.createElement('td');
    td.colSpan = row.cells.length;
    var copy = src.cloneNode(true);
    copy.removeAttribute('id');
    td.appendChild(copy);
    tr.appendChild(td);
    row.parentNode.insertBefore(tr, row.nextSibling);
    return tr;
  }
  // Capture phase so the theme's smooth-scroll handler never sees these clicks.
  document.addEventListener('click', function (event) {
    var link = event.target.closest && event.target.closest('a.bibtex-toggle');
    if (!link) return;
    event.preventDefault();
    event.stopPropagation();
    var panel = panelFor(link);
    if (!panel) return;
    panel.hidden = !panel.hidden;
    link.setAttribute('aria-expanded', String(!panel.hidden));
  }, true);
})();
</script>
